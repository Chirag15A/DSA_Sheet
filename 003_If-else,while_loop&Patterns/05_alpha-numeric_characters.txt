#include <iostream>
using namespace std;

int main() {
    char c;
    cin>>c;

    //Nested If-else
    if(c >= 'a' && c <= 'z'){
        cout<<c<<" is a lowercase character!"<<endl;
    }
    else if(c >= 'A' && c <= 'Z'){
        cout<<c<<" is a uppercase character!"<<endl;
    }
    else if(c >= '0' && c <= '9'){
        cout<<c<<" is a numeric character!"<<endl;
    }
    else{
        cout<<c<<" is not an alphanumeric character!"<<endl;
    }

    return 0;
}

Input: t
Output: t is a lowercase character!
