#include <iostream>
using namespace std;

int main() {

    int n = 4;
    int i = 0, j = 0;

    while(i < n){
        j = 0;
        while(j < i + 1){
            char c = 'A' + i;
            cout<<c<<"\t";
            j = j + 1;
        }
        cout<<endl;
        i = i + 1;
    }

    return 0;
}

Output:
A	
B	B	
C	C	C	
D	D	D	D
