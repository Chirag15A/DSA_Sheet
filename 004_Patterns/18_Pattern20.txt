#include <iostream>
using namespace std;

int main() {

    int n = 4;
    int i = 0, j = 0;

    while(i < n){
        j = 0;
        while(j < n - i ){
            cout<<"*\t";
            j = j + 1;
        }
        
        cout<<endl;
        i = i + 1;
    }

    return 0;
}

*	*	*	*	
*	*	*	
*	*	
*
