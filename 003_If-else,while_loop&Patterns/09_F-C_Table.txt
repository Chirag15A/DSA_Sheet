#include <iostream>
using namespace std;

int main() {

    int n = 10;
    int i = 1;

    while(i <= n){
        double f = 9.00/5*i + 32;
        cout<<"Celcius "<<i<<" : Fahrehniet "<<f<<endl;
        i = i + 1;
    }

    return 0;
}
