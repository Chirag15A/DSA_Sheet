#include <iostream>
using namespace std;

int main() {

    int n = 4;
    int i = 0, j = 0;
    int count = 0;

    while(i < n){
        j = 0;
        while(j < n){
            char c = 'A' + count++;
            cout<<c<<"\t";
            j = j + 1;
        }
        cout<<endl;
        i = i + 1;
    }

    return 0;
}

Output: 
A	B	C	D	
E	F	G	H	
I	J	K	L	
M	N	O	P
