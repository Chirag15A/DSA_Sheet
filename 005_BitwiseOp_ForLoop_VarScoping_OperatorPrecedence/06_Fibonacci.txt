06_Fibonacci

#include <iostream>
using namespace std;

int main() {

    int n = 20;

    int a = 0, b = 1;

    cout<<a<<endl;
    cout<<b<<endl;
    for(int i = 1; i <= n - 2; i++){
        int c = a + b;
        cout<<c<<endl;
        a = b;
        b = c;
    }
    
    return 0;
}
