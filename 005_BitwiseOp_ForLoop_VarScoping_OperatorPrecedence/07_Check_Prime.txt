07_Check_Prime

#include <iostream>
using namespace std;

int main() {

    int n = 11;
    bool flag = true;
    for(int i = 2; i < n; i++){
        if(n%i == 0){
            flag = false;
            break;
        }
    }
    if(flag) cout<<"It is prime!"<<endl;
    else cout<<"It is not prime!"<<endl;
    
    return 0;
}

Output:
It is prime!
