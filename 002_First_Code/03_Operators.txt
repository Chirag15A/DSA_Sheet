#include <iostream>
using namespace std;
	
int main() {
    int a = 5;
    int b = 7;

    //Unary Operators
    a++;
    cout<<"a: "<<a<<endl;
    b--;
    cout<<"b: "<<b<<endl;

    cout<<endl;
    //Arithmetic Operators
    cout<<"a+b: "<<a+b<<endl;
    cout<<"a-b: "<<a-b<<endl;
    cout<<"a*b: "<<a*b<<endl;
    cout<<"a/b: "<<a/b<<endl;
    cout<<"a%b: "<<a%b<<endl;
    
    return 0;
}

Output:
a: 6
b: 6

a+b: 12
a-b: 0
a*b: 36
a/b: 1
a%b: 0
