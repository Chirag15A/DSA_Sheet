02_Complement_Base10Integer

class Solution {
public:
    int bitwiseComplement(int n) {
        if(n == 0) return 1;
        int i = 0, ans = 0;
        while(n > 0){
            int bit = (n & 1);
            if(!bit) ans = ans + pow(2, i);
            i++;
            n = n >> 1;
        }
        return ans;
    }
};
