#include <iostream>
using namespace std;

int main() {
    int a = 5;
    int b = 5;

    //Nested If-else
    if(a>b){
        cout<<"A is greater!"<<endl;
    }
    else{
        if(a < b){
            cout<<"B is greater!"<<endl;
        }
        else{
            cout<<"A and B are equal!"<<endl;
        }
    }

    return 0;
}

Output:
A and B are equal!
