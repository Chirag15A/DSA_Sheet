03_Sum_of_elements

#include <bits/stdc++.h>
using namespace std;

int getSum(int arr[], int n){
    int val = 0;
    for(int i = 0; i < n; i++){
        val += arr[i];
    }
    return val;
}

int main() {
    int arr[5];
    for(int i = 0; i < 5; i++) cin>>arr[i];

    int sum = getSum(arr, 5);
    cout<<"Sum of all elements in the array is: "<<sum<<endl;
}


Console:
2 5 7 9 23
Sum of all elements in the array is: 46
