04_Power_function

#include <iostream>
using namespace std;

int power(int n, int m){
    int ans = 1;
    for(int i = 0; i < m; i++){
        ans *= n;
    }
    return ans;
}

int main() {

    int a, b;
    cout<<"Enter a: ";
    cin>>a;
    cout<<"Enter b: ";
    cin>>b;

    cout<<power(a, b)<<endl;

    
    return 0;
}
