06_Calculate_nCr

#include <iostream>
using namespace std;

int factorial(int n){
    int ans = 1;
    for(int i = 1; i <= n; i++){
        ans = ans * i;
    }
    return ans;
}

int nCr(int n, int r){
    int fact1 = factorial(n);
    int fact2 = factorial(r);
    int fact3 = factorial(n-r);
    return fact1 / (fact2 * fact3);
}

int main() {
    int n, r;
    cout<<"Enter n: ";
    cin>>n;
    cout<<"Enter r: ";
    cin>>r;

    cout<<"nCr is: "<<nCr(n, r)<<endl;
    
    return 0;
}

Console:
Enter n: 5
Enter r: 2
nCr is: 10
