#include <iostream>
using namespace std;

int main() {

    int n = 4;
    int i = 0, j = 0;

    while(i < n){
        j = 0;
        while(j < i + 1){
            char c = 'A' + n - i + j - 1;
            cout<<c<<"\t";
            j = j + 1;
        }
        cout<<endl;
        i = i + 1;
    }

    return 0;
}

Output:
D	
C	D	
B	C	D	
A	B	C	D
