#include <iostream>
using namespace std;

int main() {

    int n = 4;
    int i = 0, j = 0;

    while(i < n){
        j = 0;
        while(j < i ){
            cout<<" \t";
            j = j + 1;
        }
        j = 0;
        while(j < n - i ){
            cout<<i+1<<"\t";
            j = j + 1;
        }
        
        cout<<endl;
        i = i + 1;
    }

    return 0;
}

 	2	2	2	
 	 	3	3	
 	 	 	4
