02_Dec-Binary-Negative


#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cout << "Enter negative n: ";
    cin >> n;

    if (n >= 0) {
        cout << "Please enter a negative number";
        return 0;
    }

    long long num = -(long long)n;

    string ans = "";

    // Convert positive number to 32-bit binary
    for (int i = 0; i < 32; i++) {
        int r = num % 2;

        if (r == 0)
            ans = "1" + ans;
        else
            ans = "0" + ans;

        num /= 2;
    }

    // Add 1 to get 2's complement
    for (int i = 31; i >= 0; i--) {
        if (ans[i] == '0') {
            ans[i] = '1';
            break;
        }
        else {
            ans[i] = '0';
        }
    }

    cout << "Answer is: " << ans << endl;
}


Output:
Enter negative n: -5
Answer is: 11111111111111111111111111111011
