10_Number_of_Set_Bits

#include <iostream>
using namespace std;

int setBits(int n){
    int count = 0;

    while(n != 0){
        int bit = (n & 1);
        if(bit) count++;
        n = n >> 1;
    }

    return count;
}

int main() {
    int n;
    cout<<"Enter n: ";
    cin>>n;

    cout<<"Number of set bits is: "<<setBits(n);
    
    return 0;
}

Console:
Enter n: 10
Number of set bits is: 2
