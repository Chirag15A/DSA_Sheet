04_linear_search

#include <bits/stdc++.h>
using namespace std;

bool ifpresent(int arr[], int n, int target){
    for(int i = 0; i < n; i++){
        if(arr[i] == target) return true;
    }
    return false;
}

int main() {
    int arr[5];
    for(int i = 0; i < 5; i++) cin>>arr[i];

    cout<<"Enter element to search: ";
    int target;
    cin>>target;
    
    if(ifpresent(arr, 5, target)) cout<<"Target present."<<endl;
    else cout<<"Target not present."<<endl;
}

Console:

Target present.
