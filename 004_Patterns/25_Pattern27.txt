#include <iostream>
using namespace std;

int main() {

    int n = 5;
    int i = 0, j = 0;

    while(i < n){
        j = 0;
        while(j < n - i){
            cout<<j+1<<"\t";
            j = j + 1;
        }
        j = 0;
        while(j < i ){
            cout<<"*\t*\t";
            j = j + 1;
        }
        j = 0;
        while(j < n - i){
            cout<<n - i - j<<"\t";
            j = j + 1;
        }
        
        cout<<endl;
        i = i + 1;
    }

    return 0;
}

1	2	*	*	*	*	*	*	2	1	
1	*	*	*	*	*	*	*	*	1
