03_Count_Notes

#include <iostream>
using namespace std;

int main() {

    int n;
    cout<<"Enter amount: ";
    cin>>n;

    int temp;
    if(n >= 100) temp = 100;
    else if(n >= 50) temp = 50;
    else if(n >= 20) temp = 20;
    else temp = 1;
    

    switch(temp){
        case 100:
            if(n/100 != 0){
                cout<<n/100<<" notes of Rs 100."<<endl;
                n = n %100;
            }
        case 50:
            if(n/50 != 0){
                cout<<n/50<<" notes of Rs 50."<<endl;
                n = n % 50;
            }
        case 20:
            if(n/20 != 0){
                cout<<n/20<<" notes of Rs 20."<<endl;
                n = n % 20;
            }
        default:
            if(n/1 != 0){
                cout<<n/1<<" notes of Rs 1."<<endl;
            }
    }
    
    return 0;
}

Console:
