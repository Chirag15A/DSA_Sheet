11_nth_Fibonacci

#include <iostream>
using namespace std;

int nthFibonacci(int n){
    int a = 0;
    int b = 1;

    if(n == 1) return 0;
    if(n == 2) return 1;

    int c;
    for(int i = 2; i < n; i++){
        c = a + b;
        a = b;
        b = c;
    }

    return c;
}

int main() {
    int n;
    cout<<"Enter n: ";
    cin>>n;

    cout<<"nTh Fibonacci number is: "<<nthFibonacci(n);
    
    return 0;
}
