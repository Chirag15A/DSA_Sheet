#include <iostream>
using namespace std;

int main() {

    int n = 4;
    int i = 0, j = 0;

    while(i < n){
        j = 0;
        while(j < i + 1){
            cout<<i-j+1<<"\t";
            j = j + 1;
        }
        cout<<endl;
        i = i + 1;
    }

    return 0;
}

Output:
1	
2	1	
3	2	1	
4	3	2	1
