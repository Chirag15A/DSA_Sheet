05_Is_Even

#include <iostream>
using namespace std;

bool isEven(int n){
    return !(n%2);
}

int main() {

    int a, b;
    cout<<"Enter a: ";
    cin>>a;
    cout<<"Enter b: ";
    cin>>b;

    if(isEven(a)) cout<<a<<" is even."<<endl;
    else cout<<a<<" is odd."<<endl;

    if(isEven(b)) cout<<b<<" is even."<<endl;
    else cout<<b<<" is odd."<<endl;
    
    return 0;
}

Console:
Enter a: 5
Enter b: 56
5 is odd.
56 is even.
