01_Decimal-Binary

#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cout<<"Enter value for n: ";
    cin>>n;

    int i = 0, ans = 0;
    while(n != 0){
        int r = n % 2;
        ans = ans + pow(10, i++)*r;
        n = n / 2;
    }

    cout<<"Answer is: "<<ans<<endl;
}


Console: 
Enter value for n: 64
Answer is: 1000000
